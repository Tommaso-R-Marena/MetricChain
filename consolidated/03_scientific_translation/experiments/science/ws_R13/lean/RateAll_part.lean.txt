
/-! ## Statement (a) with the levels numbered from zero, statement (b), and the exact recursion of the cube -/

namespace ScienceR13

open MetricChain.HintRecursion

variable {θna θh : ℝ → ℝ} {a h E : ℝ}

theorem c1_pos {a h : ℝ} (hh : 0 < h) (hah : h < a) : 0 < c1 a h := by
  unfold c1
  have hκ := kappa_pos hh hah
  positivity

theorem one_add_c2_pos {a h : ℝ} (hh : 0 < h) (hah : h < a) : 0 < 1 + c2 a h := by
  have hκ := kappa_pos hh hah
  have hκd := kappa_lt_d0 hh hah
  rw [c2_eq hh hah]
  have : 1 < h / (a - h) / kappa a h := by rw [lt_div_iff₀ hκ]; linarith
  linarith

/-- **Statement (a), levels numbered from zero** (`T 0` is `T₁` of the report, as for `exactRec`). -/
theorem rate_top0 (hp : HintPair θna θh a h E) (hah : h < a) (T : ℕ → ℝ → ℝ)
    (hstep : ∀ k, StepLE θna (T k) (T (k + 1)))
    (hmono : ∀ k x, 0 < x → T (k + 1) x ≤ T k x)
    (hanti : ∀ k x y, 0 < x → x ≤ y → T k y ≤ T k x)
    (hbdd : ∀ x, 0 < x → T 0 x ≤ a) :
    ∀ (n : ℕ) (x : ℝ), 0 < x → T n x ≤ h + c1 a h / (((n + 1 : ℕ) : ℝ) + c2 a h) := by
  have key := rate_of_epochs hp.h_pos hah (fun n b => ∀ x, 0 < x → T (n - 1) x ≤ b)
    (by
      intro n b hn hP x hx
      obtain ⟨m, rfl⟩ : ∃ m, n = m + 1 := ⟨n - 1, by omega⟩
      have e : m + 1 + 1 - 1 = m + 1 := by omega
      have e' : m + 1 - 1 = m := by omega
      rw [e]
      rw [e'] at hP
      exact (hmono m x hx).trans (hP x hx))
    (fun _ _ _ hP hbb' x hx => (hP x hx).trans hbb')
    (by
      intro x hx
      exact hbdd x hx)
    (by
      intro n b hn hP hb x hx
      obtain ⟨m, rfl⟩ : ∃ m, n = m + 1 := ⟨n - 1, by omega⟩
      have e' : m + 1 - 1 = m := by omega
      rw [e'] at hP
      have e : m + 1 + ⌈h / (b - h)⌉₊ + 1 - 1 = m + ⌈h / (b - h)⌉₊ + 1 := by omega
      rw [e]
      exact epoch_of_stepLE hp T hstep hanti hb hP x hx)
  intro n x hx
  have := key (n + 1) (by omega) x hx
  simpa using this

/-- **Statement (b) of the rate, machine-checked: every accuracy.**  With the constant `L ≥ 0` of `hint_lower`
(`θh e ≥ h − L e` on `(0, E)`; `L = 1` for the cube), for `0 < g < E`, `s = g + θh g` and
`A = max{c₁ + L s, (1 + c₂)(a − h + L g)}`: for every level `n` (numbered from zero) and every `0 < y ≤ g`,
`T n y − θh y ≤ A/(n + 1 + c₂)`. -/
theorem rate_all0 (hp : HintPair θna θh a h E) (hah : h < a) {L : ℝ} (hL0 : 0 ≤ L)
    (hLow : ∀ e, 0 < e → e < E → h - L * e ≤ θh e) (T : ℕ → ℝ → ℝ)
    (hstep : ∀ k, StepLE θna (T k) (T (k + 1)))
    (hmono : ∀ k x, 0 < x → T (k + 1) x ≤ T k x)
    (hanti : ∀ k x y, 0 < x → x ≤ y → T k y ≤ T k x)
    (hbdd : ∀ x, 0 < x → T 0 x ≤ a) {g : ℝ} (hg : 0 < g) (hgE : g < E) :
    ∀ (n : ℕ) (y : ℝ), 0 < y → y ≤ g →
      T n y - θh y ≤
        max (c1 a h + L * (g + θh g)) ((1 + c2 a h) * (a - h + L * g)) / (((n + 1 : ℕ) : ℝ) + c2 a h) := by
  have hh := hp.h_pos
  have hc1 := c1_pos hh hah
  have hc2 := one_add_c2_pos hh hah
  have hθg : 0 < θh g := hp.hint_pos g hg hgE
  obtain ⟨s, hs⟩ : ∃ s : ℝ, s = g + θh g := ⟨_, rfl⟩
  have hs0 : 0 < s := by rw [hs]; linarith
  obtain ⟨A, hA⟩ : ∃ A : ℝ, A = max (c1 a h + L * s) ((1 + c2 a h) * (a - h + L * g)) := ⟨_, rfl⟩
  rw [← hs, ← hA]
  have hA1 : c1 a h + L * s ≤ A := by rw [hA]; exact le_max_left _ _
  have hA2 : (1 + c2 a h) * (a - h + L * g) ≤ A := by rw [hA]; exact le_max_right _ _
  have hApos : 0 < A := by nlinarith [mul_nonneg hL0 hs0.le]
  have htop := rate_top0 hp hah T hstep hmono hanti hbdd
  intro n
  induction n with
  | zero =>
    intro y hy hyg
    have hyE : y < E := lt_of_le_of_lt hyg hgE
    have h1 := hbdd y hy
    have h2 := hLow y hy hyE
    have h3 : L * y ≤ L * g := mul_le_mul_of_nonneg_left hyg hL0
    have hden : ((0 + 1 : ℕ) : ℝ) + c2 a h = 1 + c2 a h := by push_cast; ring
    rw [hden, le_div_iff₀ hc2]
    have h4 : (T 0 y - θh y) * (1 + c2 a h) ≤ (a - h + L * g) * (1 + c2 a h) :=
      mul_le_mul_of_nonneg_right (by linarith) hc2.le
    linarith
  | succ n ih =>
    intro y hy hyg
    have hyE : y < E := lt_of_le_of_lt hyg hgE
    have hθy : 0 < θh y := hp.hint_pos y hy hyE
    have hsum : 0 < y + θh y := by linarith
    obtain ⟨K, hK⟩ : ∃ K : ℝ, K = ((n + 1 : ℕ) : ℝ) + c2 a h := ⟨_, rfl⟩
    have hKpos : 0 < K := by
      rw [hK]; push_cast
      have : (0 : ℝ) ≤ n := Nat.cast_nonneg n
      linarith
    have hK1 : ((n + 1 + 1 : ℕ) : ℝ) + c2 a h = K + 1 := by rw [hK]; push_cast; ring
    rw [hK1]
    rw [← hK] at ih
    obtain ⟨m, hm⟩ : ∃ m : ℝ, m = A / K := ⟨_, rfl⟩
    rw [← hm] at ih
    have hm0 : 0 < m := by rw [hm]; positivity
    -- (L3) with C = y + θh y + m
    have hC : ∀ z, 0 < z → z ≤ y → z + T n z ≤ y + θh y + m := by
      intro z hz hzy
      have h1 := ih z hz (le_trans hzy hyg)
      have h2 := hp.hint_mono z y hz hzy hyE
      linarith
    have key := StepLE.at_hint_fraction hp (hstep n) (hanti n) hy hyE hC
    obtain ⟨v, hv⟩ : ∃ v : ℝ, v = T (n + 1) y - θh y := ⟨_, rfl⟩
    rw [← hv]
    -- first line: v s ≤ (s − y) m
    have hys : y + θh y ≤ s := by
      rw [hs]
      rcases eq_or_lt_of_le hyg with heq | hlt
      · rw [heq]
      · exact hp.hint_mono y g hy hyg hgE
    have hv1 : v * s ≤ (s - y) * m := by
      have e : θh y / (y + θh y) * (y + θh y + m) = θh y + θh y / (y + θh y) * m := by
        field_simp
      rw [e] at key
      have h1 : v ≤ θh y / (y + θh y) * m := by rw [hv]; linarith
      have h2 : θh y / (y + θh y) * s ≤ s - y := by
        rw [div_mul_eq_mul_div, div_le_iff₀ hsum]
        nlinarith
      calc v * s ≤ θh y / (y + θh y) * m * s := mul_le_mul_of_nonneg_right h1 hs0.le
        _ = θh y / (y + θh y) * s * m := by ring
        _ ≤ (s - y) * m := mul_le_mul_of_nonneg_right h2 hm0.le
    -- second line: v ≤ c₁/(K + 1) + L y
    have hv2 : v * (K + 1) ≤ c1 a h + (K + 1) * (L * y) := by
      have h1 := htop (n + 1) y hy
      rw [hK1] at h1
      have h2 := hLow y hy hyE
      have hK1pos : 0 < K + 1 := by linarith
      have h3 : v ≤ c1 a h / (K + 1) + L * y := by rw [hv]; linarith
      have h4 := mul_le_mul_of_nonneg_right h3 hK1pos.le
      have e : (c1 a h / (K + 1) + L * y) * (K + 1) = c1 a h + (K + 1) * (L * y) := by
        field_simp
      rw [e] at h4
      exact h4
    -- conclusion
    have hK1pos : 0 < K + 1 := by linarith
    rw [le_div_iff₀ hK1pos]
    have hv1' : K * (v * s) ≤ (s - y) * A := by
      have h1 := mul_le_mul_of_nonneg_left hv1 hKpos.le
      have e : K * ((s - y) * m) = (s - y) * A := by
        rw [hm]
        field_simp
      rw [e] at h1
      exact h1
    have hW : 0 < A + K * (L * s) := by
      have : 0 ≤ K * (L * s) := mul_nonneg hKpos.le (mul_nonneg hL0 hs0.le)
      linarith
    have T1 : v * (K + 1) * A ≤ (c1 a h + (K + 1) * (L * y)) * A := mul_le_mul_of_nonneg_right hv2 hApos.le
    have T2 : (K + 1) * L * (K * (v * s)) ≤ (K + 1) * L * ((s - y) * A) :=
      mul_le_mul_of_nonneg_left hv1' (mul_nonneg hK1pos.le hL0)
    have T3 : A * (c1 a h + L * s) ≤ A * A := mul_le_mul_of_nonneg_left hA1 hApos.le
    have hfin : v * (K + 1) * (A + K * (L * s)) ≤ A * (A + K * (L * s)) := by nlinarith
    exact le_of_mul_le_mul_right hfin hW

end ScienceR13

/-! ### The exact recursion of the cube: nothing assumed -/

namespace ScienceR13

open MetricChain.HintRecursion MetricChain.BoundedAccuracy Set

/-- **The rate for the exact level recursion of the cube** (`exactRec (cubeNa u) k` is `T_{k+1}` of the theory report):
for every slope `u > 0`, every level `k` and every accuracy rate `x > 0`,
`T_{k+1}(x) ≤ h + c₁/(k + 1 + c₂)` with `a = KL(p_u‖½)`, `h = KL(Λ₁/u‖½)`. -/
theorem cube_exactRec_rate_top {u : ℝ} (hu : 0 < u) (k : ℕ) {x : ℝ} (hx : 0 < x) :
    exactRec (cubeNa u) k x ≤
      cubeH u + c1 (cubeA u) (cubeH u) / (((k + 1 : ℕ) : ℝ) + c2 (cubeA u) (cubeH u)) := by
  have hθ : ∀ x, 0 < x → 0 ≤ cubeNa u x := fun x _ => cubeNa_nonneg u x
  have hθanti : ∀ x y, 0 < x → x ≤ y → cubeNa u y ≤ cubeNa u x := fun x y hx hxy =>
    cubeNa_antitoneOn hu (mem_Ici.mpr hx.le) (mem_Ici.mpr (hx.le.trans hxy)) hxy
  have hθC : ∀ x, 0 < x → cubeNa u x ≤ cubeA u := fun x hx => cubeNa_le_cubeA hu hx.le
  have P := fun k => exactRec_props hθ hθanti hθC k
  have hah : cubeH u < cubeA u := by
    have h1 := cubeHint_lt_cubeNa hu le_rfl (cubeEstar_pos hu)
    rw [cubeHint_zero, cubeNa_zero hu] at h1
    exact h1
  exact rate_top0 (cubePair hu) hah (exactRec (cubeNa u))
    (fun k => stepLE_exact hθ (P k).1)
    (fun k x _ => min_le_left _ _)
    (fun k x y hx hxy => (P k).2.2 x y hx hxy)
    (fun x hx => (P 0).2.1 x hx) k x hx

/-- **The rate at every accuracy for the exact level recursion of the cube** (`L = 1`): for `0 < g < ε*`,
`T_{k+1}(g) − θ_hint(g) ≤ A(g)/(k + 1 + c₂)`, `A(g) = max{c₁ + g + θ_hint(g), (1 + c₂)(a − h + g)}`. -/
theorem cube_exactRec_rate_all {u : ℝ} (hu : 0 < u) (k : ℕ) {g : ℝ} (hg : 0 < g) (hgE : g < cubeEstar u) :
    exactRec (cubeNa u) k g - cubeHint u g ≤
      max (c1 (cubeA u) (cubeH u) + (g + cubeHint u g)) ((1 + c2 (cubeA u) (cubeH u)) * (cubeA u - cubeH u + g)) /
        (((k + 1 : ℕ) : ℝ) + c2 (cubeA u) (cubeH u)) := by
  have hθ : ∀ x, 0 < x → 0 ≤ cubeNa u x := fun x _ => cubeNa_nonneg u x
  have hθanti : ∀ x y, 0 < x → x ≤ y → cubeNa u y ≤ cubeNa u x := fun x y hx hxy =>
    cubeNa_antitoneOn hu (mem_Ici.mpr hx.le) (mem_Ici.mpr (hx.le.trans hxy)) hxy
  have hθC : ∀ x, 0 < x → cubeNa u x ≤ cubeA u := fun x hx => cubeNa_le_cubeA hu hx.le
  have P := fun k => exactRec_props hθ hθanti hθC k
  have hah : cubeH u < cubeA u := by
    have h1 := cubeHint_lt_cubeNa hu le_rfl (cubeEstar_pos hu)
    rw [cubeHint_zero, cubeNa_zero hu] at h1
    exact h1
  have hLow : ∀ e, 0 < e → e < cubeEstar u → cubeH u - 1 * e ≤ cubeHint u e := by
    intro e he0 heE
    have h1 := cubeHint_add_mono hu le_rfl he0.le heE.le
    rw [zero_add, cubeHint_zero] at h1
    linarith
  have key := rate_all0 (cubePair hu) hah (L := 1) zero_le_one hLow (exactRec (cubeNa u))
    (fun k => stepLE_exact hθ (P k).1)
    (fun k x _ => min_le_left _ _)
    (fun k x y hx hxy => (P k).2.2 x y hx hxy)
    (fun x hx => (P 0).2.1 x hx) hg hgE k g hg le_rfl
  simpa using key

end ScienceR13
