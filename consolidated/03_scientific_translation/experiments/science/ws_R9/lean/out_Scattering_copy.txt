sinc_sqrt_lipschitz : ∀ (q : ℝ) {s s' : ℝ},
  0 ≤ s → 0 ≤ s' → |Real.sinc (q * √s) - Real.sinc (q * √s')| ≤ q ^ 2 / 6 * |s - s'|
@debye_sub_le : ∀ {ι : Type u_1} [inst : Fintype ι] (N : ℕ) (q : ℝ) (s s' : ι → ℝ),
  (∀ (k : ι), 0 ≤ s k) →
    (∀ (k : ι), 0 ≤ s' k) → ∀ (σ : Equiv.Perm ι), |debye N q s - debye N q s'| ≤ q ^ 2 / 3 * ∑ k, |s k - s' (σ k)|
@debye_eq_of_matching : ∀ {ι : Type u_1} [inst : Fintype ι] (N : ℕ) (q : ℝ) (s s' : ι → ℝ),
  (∀ (k : ι), 0 ≤ s k) →
    (∀ (k : ι), 0 ≤ s' k) → ∀ (σ : Equiv.Perm ι), (∀ (k : ι), s k = s' (σ k)) → debye N q s = debye N q s'
