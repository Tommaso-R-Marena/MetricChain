/-- An instance of my own: four configurations, three observables
`φ₀ = (1,0,1,2)`, `φ₁ = (0,1,1,1)`, `φ₂ = (0,0,1,3)`, support `{0,1}`, certificate `λ = (0,0,1)`.
Every ensemble whose data are `t` times those of `μ = (1/3, 2/3, 0, 0)`, for ANY real `t`
(no positivity of `t` is assumed), equals `μ`. -/

    ν = fun x => if x = 0 then (1/3 : ℝ) else if x = 1 then 2/3 else 0 := by

/-- The factor is forced: under the hypotheses of the theorem, if some datum of `μ` is non-zero then `t = 1`.
(So "any real multiple" in the report's sentence is accurate, and for `t ≠ 1` the hypothesis `hdata` is
unsatisfiable when the data of `μ` are not all zero.) -/
theorem er4_factor_forced {X : Type*} [Fintype X] {ι : Type*} [Fintype ι]
    (Φ : ι → X → ℝ) (μ ν : X → ℝ) (hμ : IsEnsemble μ) (hν : IsEnsemble ν)
    (S : Finset X) (hsupp : ∀ x, x ∉ S → μ x = 0)
    (lam : ι → ℝ)
    (hzero : ∀ x ∈ S, energy lam Φ x = 0)
    (hpos : ∀ x, x ∉ S → 0 < energy lam Φ x)
    (hindep : ∀ h : X → ℝ, (∀ x, x ∉ S → h x = 0) →
        (∀ i, ∑ x, h x * Φ i x = 0) → ∀ x, h x = 0)
    (t : ℝ) (hdata : ∀ i, mean ν (Φ i) = t * mean μ (Φ i))
    (i₀ : ι) (hne : mean μ (Φ i₀) ≠ 0) : t = 1 := by
  have hνμ : ν = μ := eq_of_homogeneous_certificate Φ μ ν hμ hν S hsupp lam hzero hpos hindep t hdata
  have h := hdata i₀
  rw [hνμ] at h
  have : (1 - t) * mean μ (Φ i₀) = 0 := by linarith
  rcases mul_eq_zero.mp this with h1 | h1
  · linarith
  · exact absurd h1 hne
